483	interface WatchlistDropdownProps {
484	  selected: WatchStatus | null
485	  onSelect: (s: WatchStatus) => void
486	  /** Invoked to drop this title from the watchlist entirely. */
487	  onRemove: () => void
488	  /** True when the user has actually watched all episodes (auto-completion). */
489	  completedByCompletion: boolean
490	}
491	
492	/**
493	 * Watchlist status dropdown (Watching / Plan to Watch / Completed / On Hold /
494	 * Dropped). Each option carries a color-coded icon; "Completed" is a locked,
495	 * earned state â€” it can only be reached by watching every episode, never by
496	 * clicking it.
497	 */
498	function WatchlistDropdown({ selected, onSelect, onRemove, completedByCompletion }: WatchlistDropdownProps) {
499	  const { t } = useI18n()
500	  const [open, setOpen] = useState(false)
501	  const ref = useRef<HTMLDivElement>(null)
502	
503	  const options = [
504	    { id: 'watching', label: t('detail.statusWatching'), Icon: ListIcon, color: 'text-teal-400', locked: false },
505	    { id: 'plan', label: t('detail.statusPlanToWatch'), Icon: CalendarPlusIcon, color: 'text-sky-400', locked: false },
506	    { id: 'completed', label: t('detail.statusCompleted'), Icon: CheckIcon, color: 'text-emerald-400', locked: true },
507	    { id: 'onhold', label: t('detail.statusOnHold'), Icon: PauseCircleIcon, color: 'text-amber-400', locked: false },
508	    { id: 'dropped', label: t('detail.statusDropped'), Icon: XCircleIcon, color: 'text-red-400', locked: false },
509	  ] as const
510	
511	  useEffect(() => {
512	    if (!open) return
513	    const onClick = (e: MouseEvent) => {
514	      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false)
515	    }
516	    document.addEventListener('mousedown', onClick)
517	    return () => document.removeEventListener('mousedown', onClick)
518	  }, [open])
519	
520	  const current = options.find((o) => o.id === selected)
521	  const isCompletedSelected = current?.id === 'completed' && completedByCompletion
522	
523	  return (
524	    <div ref={ref} className="relative shrink-0">
525	      <button
526	        type="button"
527	        onClick={() => setOpen((o) => !o)}
528	        className={cx(
529	          'btn-ghost py-3',
530	          current &&
531	            (isCompletedSelected
532	              ? 'border-emerald-500/60 text-emerald-300'
533	              : 'border-teal-500/60 text-teal-300'),
534	        )}
535	      >
536	        {current ? (
537	          <current.Icon className={cx('h-5 w-5 shrink-0', current.color)} />
538	        ) : (
539	          <ListIcon className="h-5 w-5 shrink-0" />
540	        )}
541	        <span>
542	          {current?.label ?? t('detail.addToWatchlist')}
543	          {isCompletedSelected && <LockIcon className="ms-1 inline h-3.5 w-3.5 opacity-70" />}
544	        </span>
545	        <ChevronDown className={cx('h-4 w-4 shrink-0 transition', open && 'rotate-180')} />
546	      </button>
547	      {open && (
548	        <div className="absolute start-0 top-full z-30 mt-2 w-max min-w-55 overflow-hidden rounded-xl border border-night-600/80 bg-night-850 py-1 shadow-card">
549	          {options.map(({ id, label, Icon, color, locked }) => (
550	            <button
551	              key={id}
552	              type="button"
553	              onClick={() => {
554	                if (locked) return
555	                onSelect(id)
556	                setOpen(false)
557	              }}
558	              aria-disabled={locked}
559	              className={cx(
