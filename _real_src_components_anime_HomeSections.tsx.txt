12	/**
13	 * ContinueWatching â€” prominent carousel on the home page listing the anime the
14	 * user most recently opened episodes for, with a direct "resume" button back to
15	 * the exact episode (#10).
16	 */
17	export function ContinueWatching() {
18	  const { t } = useI18n()
19	  const { continueWatching, removeContinue } = useLibrary()
20	
21	  if (continueWatching.length === 0) return null
22	
23	  return (
24	    <Carousel
25	      title={t('home.continueWatching')}
26	      subtitle={t('home.continueWatchingSub')}
27	      viewAllHref="/history"
28	    >
29	      {continueWatching.map((item) => (
30	        <CarouselItem key={item.animeId}>
31	          <ContinueCard
32	            coverImage={item.coverImage}
33	            title={item.title}
34	            episode={item.episode}
35	            totalEpisodes={item.totalEpisodes ?? null}
36	            to={`/watch/${item.animeId}/${item.episode}`}
37	            onRemove={() => removeContinue(item.animeId)}
38	          />
39	        </CarouselItem>
40	      ))}
41	    </Carousel>
42	  )
43	}

45	function ContinueCard({
46	  coverImage,
47	  title,
48	  episode,
49	  totalEpisodes,
50	  to,
51	  onRemove,
52	}: {
53	  coverImage: string
54	  title: string
55	  episode: number
56	  totalEpisodes: number | null
57	  to: string
58	  onRemove: () => void
59	}) {
60	  const { t } = useI18n()
61	  const episodeLabel =
62	    totalEpisodes != null
63	      ? `${t('home.episodeShort', { n: episode })} / ${totalEpisodes}`
64	      : t('home.episodeShort', { n: episode })
65	
66	  return (
67	    <div className="group relative">
68	      <Link
69	        to={to}
70	        className="block overflow-hidden rounded-2xl border border-night-700/70 bg-night-850 transition group-hover:-translate-y-1 group-hover:border-slate-400/60 group-hover:shadow-glow-accent"
71	      >
72	        <div className="relative aspect-[16/10] overflow-hidden">
73	          {coverImage ? (
74	            <img
75	              src={coverImage}
76	              alt={title}
77	              loading="lazy"
78	              decoding="async"
79	              className="h-full w-full object-cover transition duration-500 group-hover:scale-105"
80	            />
81	          ) : (
82	            <div className="flex h-full w-full items-center justify-center bg-night-800 text-slate-600">
83	              ?
84	            </div>
85	          )}

87	          {/* Play affordance */}

106	        type="button"
107	        onClick={onRemove}
108	        aria-label={t('history.remove')}

112	      </button>
113	    </div>
114	  )
115	}

117	/**
118	 * TasteSection â€” "Animes Based on Your Taste". If the user has watch history we
119	 * extract their most-watched genres and fetch recommendations for them; if not,
120	 * we show a curated "Must Watch" (top popularity) row (#10).
121	 */
122	export function TasteSection() {
123	  const { t } = useI18n()
124	  const { history } = useLibrary()
125	  const topGenres = topGenresFromHistory(3)
126	  const hasHistory = history.length > 0 && topGenres.length > 0
127	  const primaryGenre = hasHistory ? topGenres[0] : null
128	
129	  const recs = useAsync(
130	    (s): Promise<AnimeSummary[]> =>
131	      primaryGenre
132	        ? anilist
133	            .searchAnime({ genre: primaryGenre, order_by: 'popularity', limit: 24, sfw: true }, s)
134	            .then((r) => r.data)
135	        : anilist.getTopAnime('bypopularity', 1, s).then((r) => r.data),
136	    [primaryGenre],
137	  )
138	
139	  const title = hasHistory ? t('home.basedOnTaste') : t('home.mustWatch')
140	  const subtitle = hasHistory ? t('home.basedOnTasteSub') : t('home.mustWatchSub')
141	
142	  return (
143	    <Carousel title={title} subtitle={subtitle} viewAllHref="/catalog?sort=popularity">
144	      {recs.loading && <AnimeCardSkeleton count={7} />}
145	      {recs.error && <p className="px-1 py-4 text-sm text-red-400">{recs.error}</p>}
146	      {recs.data?.map((a) => (
147	        <CarouselItem key={a.mal_id}>
148	          <AnimeCard anime={a} />
149	        </CarouselItem>
150	      ))}
151	    </Carousel>
152	  )
153	}
