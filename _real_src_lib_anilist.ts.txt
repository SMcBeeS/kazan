44	let lastRequestAt = 0
45	const queue: Array<() => void> = []
46	let draining = false
47	
48	const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms))
49	
50	async function drain() {
51	  if (draining) return
52	  draining = true
53	  while (queue.length > 0) {
54	    const now = Date.now()
55	    const wait = Math.max(0, MIN_INTERVAL - (now - lastRequestAt))
56	    if (wait > 0) await sleep(wait)
57	    lastRequestAt = Date.now()
58	    const next = queue.shift()
59	    next?.()
60	  }
61	  draining = false
62	}
63	
64	function schedule<T>(task: () => Promise<T>): Promise<T> {
65	  return new Promise<T>((resolve, reject) => {
66	    queue.push(() => {
67	      task().then(resolve, reject)
68	    })
69	    void drain()
70	  })
71	}

75	interface CacheEntry {
76	  value: unknown
77	  expires: number
78	}
79	
80	const memoryCache = new Map<string, CacheEntry>()
81	const inflight = new Map<string, Promise<unknown>>()
82	
83	function readStorage(key: string, allowStale = false): unknown | undefined {
84	  try {
85	    const raw = localStorage.getItem(key)
86	    if (!raw) return undefined
87	    const parsed = JSON.parse(raw) as CacheEntry
88	    if (!allowStale && Date.now() > parsed.expires) {
89	      return undefined
90	    }
91	    return parsed.value
92	  } catch {
93	    return undefined
94	  }
95	}
96	
97	function writeStorage(key: string, value: unknown, ttl: number) {
98	  try {
99	    const entry: CacheEntry = { value, expires: Date.now() + ttl }
100	    localStorage.setItem(key, JSON.stringify(entry))
101	  } catch {
102	    /* non-fatal */
103	  }
104	}
105	
106	function cacheGet<T>(key: string, allowStale = false): T | undefined {
107	  const mem = memoryCache.get(key)
108	  if (mem && (allowStale || Date.now() < mem.expires)) return mem.value as T
109	  const stored = readStorage(key, allowStale)
110	  if (stored !== undefined) {
111	    memoryCache.set(key, { value: stored, expires: Date.now() + TTL.default })
112	    return stored as T
113	  }
114	  return undefined
115	}
116	
117	function cacheSet<T>(key: string, value: T, ttl: number) {
118	  memoryCache.set(key, { value, expires: Date.now() + ttl })
119	  writeStorage(key, value, ttl)
120	}

124	interface GqlOptions {
125	  variables?: Record<string, unknown>
126	  ttl?: number
127	  signal?: AbortSignal
128	}
129	
130	function friendlyNetworkMessage(err: unknown): string {
131	  if (err instanceof DOMException && err.name === 'AbortError') return 'aborted'
132	  return 'Could not reach AniList right now. Please try again in a moment.'
133	}
134	
135	async function gql<T>(query: string, opts: GqlOptions = {}): Promise<T> {
136	  const { variables, ttl = TTL.default, signal } = opts
137	
138	  let hash = 0x811c9dc5
139	  const rawKey = `${query}::${JSON.stringify(variables ?? {})}`
140	  for (let i = 0; i < rawKey.length; i++) {
141	    hash ^= rawKey.charCodeAt(i)
142	    hash = Math.imul(hash, 0x01000193)
143	  }
144	  const cacheKey = `kazan_cache_${(hash >>> 0).toString(36)}`
145	
146	  const cached = cacheGet<T>(cacheKey, false)
147	  if (cached !== undefined) return cached
148	
149	  const existing = inflight.get(cacheKey)
150	  if (existing) return existing as Promise<T>
151	
152	  const requestPromise = (async (): Promise<T> => {
153	    let attempt = 0
154	    while (true) {
155	      try {
156	        return await schedule(async () => {
157	          const res = await fetch(BASE_URL, {
158	            method: 'POST',
159	            headers: {
160	              'Content-Type': 'application/json',
161	              Accept: 'application/json',
162	            },
163	            body: JSON.stringify({ query, variables }),
164	            signal,
165	          })
166	          if (res.status === 429) throw new AniListError('rate-limited', 429)
167	          if (!res.ok) throw new AniListError(`AniList responded with ${res.status}`, res.status)
168	          const json = (await res.json()) as { data?: T; errors?: { message: string }[] }
169	          if (json.errors && json.errors.length > 0) {
170	            throw new AniListError(json.errors[0]?.message ?? 'GraphQL error', res.status)
171	          }
172	          if (!json.data) throw new AniListError('Empty response', res.status)
173	          cacheSet(cacheKey, json.data, ttl)
174	          return json.data
175	        })
176	      } catch (err) {
177	        if (signal?.aborted || (err instanceof DOMException && err.name === 'AbortError')) {
178	          throw new AniListError('aborted', 0)
179	        }
180	        const status = err instanceof AniListError ? err.status : 0
181	        attempt += 1
182	
183	        const retryable = status === 0 || status === 429 || (status >= 500 && status < 600)
184	        if (retryable && attempt <= MAX_RETRIES) {
185	          await sleep(1200 * attempt)
186	          continue
187	        }
188	
189	        const stale = cacheGet<T>(cacheKey, true)
190	        if (stale !== undefined) return stale
191	
192	        if (err instanceof AniListError) throw err
193	        throw new AniListError(friendlyNetworkMessage(err), 0)
194	      }
195	    }
196	  })()
197	
198	  inflight.set(cacheKey, requestPromise)
199	  try {
200	    return await requestPromise
201	  } finally {
202	    inflight.delete(cacheKey)
203	  }
204	}

368	const pad2 = (n: number): string => String(n).padStart(2, '0')

370	function fuzzyToIso(d?: AniListFuzzyDate | null): string | null {
371	  if (!d || !d.year) return null
372	  return `${d.year}-${pad2(d.month ?? 1)}-${pad2(d.day ?? 1)}`
373	}
374	
375	function fuzzyToString(d?: AniListFuzzyDate | null): string {
376	  if (!d || !d.year) return ''
377	  const month = d.month ? MONTHS[d.month - 1] : ''
378	  if (month && d.day) return `${month} ${d.day}, ${d.year}`
379	  if (month) return `${month} ${d.year}`
380	  return String(d.year)
381	}

383	function mapAired(start?: AniListFuzzyDate | null, end?: AniListFuzzyDate | null): JikanAired {
384	  const fromStr = fuzzyToString(start)
385	  const toStr = fuzzyToString(end)
386	  let string: string
387	  if (fromStr && toStr) string = `${fromStr} to ${toStr}`
388	  else if (fromStr) string = `${fromStr} to ?`
389	  else string = toStr || 'â€”'
390	  const toProp = (d?: AniListFuzzyDate | null): JikanAiredDate => ({
391	    day: d?.day ?? null,
392	    month: d?.month ?? null,
393	    year: d?.year ?? null,
394	  })
395	  return {
396	    from: fuzzyToIso(start),
397	    to: fuzzyToIso(end),
398	    string,
399	    prop: { from: toProp(start), to: toProp(end) },
400	  }
401	}

429	  if (!html) return null

432	    .replace(/<\/p>/gi, '\n\n')

599	function buildSearchQuery(f: SearchFilters, selection = SEARCH_MEDIA_SELECTION): string {
600	  const decls: string[] = ['$page: Int', '$perPage: Int']
601	  const args: string[] = ['type: ANIME']
602	  if (f.search != null && f.search !== '') {
603	    decls.push('$search: String')
604	    args.push('search: $search')
605	  }
606	  if (f.sort != null) {
607	    decls.push('$sort: [MediaSort]')
608	    args.push('sort: $sort')
609	  }
610	  if (f.status != null) {
611	    decls.push('$status: MediaStatus')
612	    args.push('status: $status')
613	  }
614	  if (f.format != null && f.format.length > 0) {
615	    decls.push('$format_in: [MediaFormat]')
616	    args.push('format_in: $format_in')
617	  }
618	  if (f.genre != null && f.genre.length > 0) {
619	    decls.push('$genre_in: [String]')
620	    args.push('genre_in: $genre_in')
621	  }
622	  if (f.tags != null && f.tags.length > 0) {
623	    decls.push('$tag_in: [String]')
624	    args.push('tag_in: $tag_in')
625	  }
626	  if (f.season != null) {
627	    decls.push('$season: MediaSeason')
628	    args.push('season: $season')
629	  }
630	  if (f.seasonYear != null) {
631	    decls.push('$seasonYear: Int')
632	    args.push('seasonYear: $seasonYear')
633	  }
634	  if (f.minScore != null) {
635	    decls.push('$minScore: Int')
636	    args.push('averageScore_greater: $minScore')
637	  }
638	  if (f.isAdult != null) {
639	    decls.push('$isAdult: Boolean')
640	    args.push('isAdult: $isAdult')
641	  }
642	  const body = selection.replace('ARGS_PLACEHOLDER', args.join(', '))
643	  return `query (${decls.join(', ')}) {\n${body}\n}`
644	}

646	function searchVariables(f: SearchFilters, page: number, perPage: number): Record<string, unknown> {
647	  const variables: Record<string, unknown> = { page, perPage }
648	  if (f.search != null && f.search !== '') variables.search = f.search
649	  if (f.sort != null) variables.sort = f.sort
650	  if (f.status != null) variables.status = f.status
651	  if (f.format != null && f.format.length > 0) variables.format_in = f.format
652	  if (f.genre != null && f.genre.length > 0) variables.genre_in = f.genre

690	async function resolveTruePageInfo(
691	  f: SearchFilters,
692	  page: number,
693	  perPage: number,
694	  media: unknown[],
695	  raw: PageInfo,
696	  signal?: AbortSignal,
697	): Promise<PageInfo> {
698	  const withCurrent = (info: PageInfo): PageInfo => ({ ...info, currentPage: page, perPage })
699	
700	  if (media.length > 0 && !raw.hasNextPage) return raw
701	  if (page === 1 && media.length === 0) return raw
702	
703	  let probes = 0
704	  const probe = async (p: number): Promise<ProbePage | null> => {
705	    if (probes >= MAX_PAGE_PROBES || p < 1 || p > MAX_SEARCH_PAGES) return null
706	    probes += 1
707	    return probeSearchPage(f, p, perPage, signal)
708	  }
709	
710	  let lo: number
711	  let hi: number
712	
713	  if (media.length > 0) {
714	    lo = page
715	    let emptyAt: number | null = null
716	    for (let k = page * 2; k <= MAX_SEARCH_PAGES; k *= 2) {
717	      const pr = await probe(k)
718	      if (!pr) return raw
719	      if (pr.count === 0) {
720	        emptyAt = k
721	        break
722	      }
723	      if (!pr.pageInfo.hasNextPage) return withCurrent(pr.pageInfo)
724	      lo = k
725	    }
726	    if (emptyAt === null) {
727	      const edge = await probe(MAX_SEARCH_PAGES)
728	      if (edge && edge.count > 0 && !edge.pageInfo.hasNextPage) return withCurrent(edge.pageInfo)
729	      return raw
730	    }
731	    hi = emptyAt
732	  } else {
733	    hi = page
734	    const first = await probe(1)
735	    if (!first) return raw
736	    if (first.count === 0) {
737	      return withCurrent({ total: 0, currentPage: page, lastPage: 1, hasNextPage: false, perPage })
738	    }
739	    if (!first.pageInfo.hasNextPage) return withCurrent(first.pageInfo)
740	    lo = 1
741	  }
742	
743	  while (hi - lo > 1) {
744	    const mid = (lo + hi) >> 1
745	    const pr = await probe(mid)
746	    if (!pr) return raw
747	    if (pr.count === 0) {
748	      hi = mid
749	      continue
750	    }
751	    if (!pr.pageInfo.hasNextPage) return withCurrent(pr.pageInfo)
752	    lo = mid
753	  }
754	
755	  const finalPr = await probe(lo)
756	  const finalCount = finalPr ? finalPr.count : perPage
757	
758	  return withCurrent({
759	    total: (lo - 1) * perPage + finalCount,
760	    currentPage: page,
761	    lastPage: lo,
762	    hasNextPage: lo > page,
763	    perPage,
764	  })
765	}

767	function runSearch(
768	  filters: SearchFilters,
769	  page: number,
770	  perPage: number,
771	  signal?: AbortSignal,
772	  ttl?: number,
773	  verifyPageInfo = false,
774	): Promise<Paginated<AnimeSummary>> {
775	  interface Resp {
776	    Page: { pageInfo: PageInfo; media: AniListMedia[] }
777	  }
778	  const query = buildSearchQuery(filters)
779	  const variables = searchVariables(filters, page, perPage)
780	  return gql<Resp>(query, { variables, ttl, signal }).then(async (r) => {
781	    const media = r.Page.media ?? []

783	      ? await resolveTruePageInfo(filters, page, perPage, media, r.Page.pageInfo, signal)
784	      : r.Page.pageInfo
785	    return toPaginated(pageInfo, media)
786	  })
787	}

900	export function currentSeason(): { season: string; year: number } {
901	  const now = new Date()
902	  const month = now.getMonth()
903	  const year = now.getFullYear()
904	  let season: string
905	  if (month <= 1 || month === 11) season = 'WINTER'
906	  else if (month <= 4) season = 'SPRING'
907	  else if (month <= 7) season = 'SUMMER'
908	  else season = 'FALL'
909	  return { season, year }

1308	/**
1309	 * fetchAnimeMetadata â€” batch-resolve AniList ids into lightweight display
1310	 * summaries (id + best title + large cover). Used by the profile list views to
1311	 * replace the raw "Anime #<id>" placeholders with real titles and posters.
1312	 *
1313	 * Behaviour:
1314	 *   - De-duplicates the input list and drops non-positive / non-integer ids.
1315	 *   - Returns `[]` for an empty input.
1316	 *   - Never throws: a failed/blocked request degrades to `null` so the caller
1317	 *     can still fall back to its `#<id>` labels.
1318	 */
1319	export async function fetchAnimeMetadata(
1320	  idList: number[],
1321	  signal?: AbortSignal,
1322	): Promise<AnimeMeta[] | null> {
1323	  const ids = [...new Set(idList.map((n) => Number(n)).filter((n) => Number.isFinite(n) && n > 0))]
1324	  if (ids.length === 0) return []
1325	
1326	  const query = `
1327	    query ($ids: [Int!]!) {
1328	      Page {
1329	        media(id_in: $ids, type: ANIME) {
1330	          id
1331	          title { romaji english native }
1332	          coverImage { large }
1333	        }
1334	      }
1335	    }
1336	  `
1337	  interface Resp {
1338	    Page: {
1339	      media: { id: number; title: AniListTitle; coverImage: AniListCoverImage }[] | null
1340	    } | null
1341	  }
1342	  try {
1343	    const r = await gql<Resp>(query, { variables: { ids }, signal })
1344	    const media = r.Page?.media ?? []
1345	    return media.map((m) => ({
1346	      id: m.id,
1347	      title: m.title.english || m.title.romaji || m.title.native || '',
1348	      coverImage: m.coverImage?.large ?? null,
1349	    }))
1350	  } catch {
1351	    return null
1352	  }
