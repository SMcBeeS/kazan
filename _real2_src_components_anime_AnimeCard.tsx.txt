1	import { Link } from 'react-router-dom'
2	import type { AnimeSummary } from '../../lib/types'
3	import { coverImage, cx, formatLabel, formatRating, formatScore, isMovie } from '../../lib/utils'
4	import { useI18n } from '../../context/LanguageContext'
5	import { translateGenre } from '../../i18n'
6	import { useLibrary } from '../../context/LibraryContext'
7	import { HeartIcon, PlayIcon, StarIcon } from '../icons'
8	import { Skeleton } from '../ui/Feedback'
9	
10	/**
11	 * AnimeCard â€” dashboard-style poster card: rounded poster, star score badge,
12	 * optional rank badge, and bottom metadata (title, genre line).
13	 */
14	export function AnimeCard({
15	  anime,
16	  eager,
17	  rank,
18	}: {
19	  anime: AnimeSummary
20	  eager?: boolean
21	  rank?: number
22	}) {
23	  const { t, lang } = useI18n()
24	  const { isFavorite, toggleFavorite } = useLibrary()
25	  const fav = isFavorite(anime.mal_id)
26	  const img = coverImage(anime)
27	  // Movies are single runs â€” an episode count is meaningless there, so the
28	  // card shows the total runtime instead (or nothing if unknown).
29	  const isMovieAnime = anime.format === 'MOVIE' || isMovie(anime.type)
30	  const movieRuntime = isMovieAnime && anime.duration ? anime.duration.replace(/\s*per ep$/i, '') : null
31	
32	  return (
33	    <div className="group relative">
34	      <Link
35	        to={`/anime/${anime.mal_id}`}
36	        className="block focus-visible:outline-none"
37	        aria-label={anime.title}
38	      >
39	        <div className="relative aspect-[2/3] overflow-hidden rounded-xl border border-night-700/70 bg-night-850 transition duration-300 group-hover:-translate-y-1.5 group-hover:border-slate-400/60 group-hover:shadow-glow-accent">
40	          {img ? (
41	            <img
42	              src={img}
43	              alt={anime.title}
44	              loading={eager ? 'eager' : 'lazy'}
45	              decoding="async"
46	              className="h-full w-full object-cover transition duration-500 group-hover:scale-105"
47	            />
48	          ) : (
49	            <div className="flex h-full w-full items-center justify-center text-slate-600">
50	              <span className="text-3xl">?</span>
51	            </div>
52	          )}
53	
54	          {/*
55	            Bottom scrim only â€” a single, subtle gradient that guarantees title
56	            legibility without the heavy always-on dark overlay or stacked
57	            shadows that previously washed out the poster (#2).
58	          */}
59	          <div className="pointer-events-none absolute inset-x-0 bottom-0 h-1/2 bg-gradient-to-t from-night-950/85 via-night-950/20 to-transparent" />
60	
61	          {anime.score != null && (
62	            <div className="absolute start-2 top-2 flex items-center gap-1 rounded-full bg-night-950/80 px-2 py-1 text-xs font-bold text-gold-400 backdrop-blur-sm">
63	              <StarIcon filled className="h-3.5 w-3.5" />
64	              {formatScore(anime.score)}
65	            </div>
66	          )}
67	
68	          {rank != null && (
69	            <div className="absolute end-2 top-2 flex h-7 min-w-7 items-center justify-center rounded-full bg-night-900/90 px-1.5 text-xs font-bold text-white ring-1 ring-white/20">
70	              {rank}
71	            </div>
72	          )}
73	
74	          {anime.type && rank == null && (
75	            <span className="absolute end-2 top-2 rounded-full bg-night-900/90 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-slate-100 ring-1 ring-white/15 backdrop-blur-sm">
76	              {formatLabel(anime.type, lang)}
77	            </span>
78	          )}
79	
80	          {/* Play affordance on hover */}
81	          <div className="absolute inset-0 flex items-center justify-center opacity-0 transition duration-300 group-hover:opacity-100">
82	            <span className="flex h-12 w-12 items-center justify-center rounded-full bg-white/90 text-night-900 shadow-glow-accent">
83	              <PlayIcon className="ms-0.5 h-5 w-5" />
84	            </span>
85	          </div>
86	
87	          {/* Title overlay */}
88	          <div className="absolute inset-x-0 bottom-0 p-2.5">
89	            <h3 className="line-clamp-1 text-sm font-bold text-white text-shadow-lg">
90	              {anime.title_english || anime.title}
91	            </h3>
92	            <div className="mt-1 flex flex-wrap items-center gap-1.5">
93	              {!isMovieAnime && anime.episodes != null && (
94	                <span className="inline-flex items-center gap-1 rounded-full bg-night-950/70 px-1.5 py-0.5 text-[10px] font-semibold text-slate-200 backdrop-blur-sm">
95	                  {anime.episodes} {t('detail.episodes')}
96	                </span>
97	              )}
98	              {movieRuntime && (
99	                <span className="inline-flex items-center gap-1 rounded-full bg-night-950/70 px-1.5 py-0.5 text-[10px] font-semibold text-slate-200 backdrop-blur-sm">
100	                  {movieRuntime}
101	                </span>
102	              )}
103	              {anime.rating && (
104	                <span className="inline-flex items-center rounded-full bg-night-950/70 px-1.5 py-0.5 text-[10px] font-semibold text-slate-300 backdrop-blur-sm">
105	                  {formatRating(anime.rating)}
106	                </span>
107	              )}
108	            </div>
109	          </div>
110	        </div>
111	      </Link>
112	
113	      {/* Bottom metadata: title, genre line, star score */}
114	      <div className="mt-2 px-0.5">
115	        <h3 className="line-clamp-1 text-sm font-bold text-white">
116	          {anime.title_english || anime.title}
117	        </h3>
118	        <div className="mt-1 flex items-center justify-between gap-2 text-[11px]">
119	          <span className="min-w-0 flex-1 truncate font-medium text-slate-400">
120	            {(anime.genres ?? []).slice(0, 2).map((g) => translateGenre(g.name, lang)).join(' â€¢ ') ||
121	              formatLabel(anime.type ?? '', lang)}
122	          </span>
123	          {anime.score != null && (
124	            <span className="flex shrink-0 items-center gap-1 font-bold text-gold-400">
125	              <StarIcon filled className="h-3 w-3" />
126	              {formatScore(anime.score)}
127	            </span>
128	          )}
129	        </div>
130	      </div>
131	
132	      {/* Favorite toggle (outside link so it doesn't navigate) */}
133	      <button
134	        type="button"
135	        onClick={() => toggleFavorite(anime)}
136	        aria-label={fav ? t('detail.removeFavorite') : t('detail.addFavorite')}
137	        aria-pressed={fav}
138	        className={cx(
139	          'absolute end-2 bottom-[62px] z-10 flex h-9 w-9 items-center justify-center rounded-full border backdrop-blur-sm transition',
140	          fav
141	            ? 'border-white/50 bg-white/10 text-white hover:bg-white/20'
142	            : 'border-night-600/80 bg-night-900/70 text-slate-300 opacity-0 hover:border-white/50 hover:text-white group-hover:opacity-100 focus-visible:opacity-100',
143	        )}
144	      >
145	        <HeartIcon filled={fav} className="h-4 w-4" />
146	      </button>
147	    </div>
148	  )
149	}
150	
151	/** Grid of skeleton cards used while loading lists. */
152	export function AnimeCardSkeleton({ count = 12 }: { count?: number }) {
153	  return (
154	    <>
155	      {Array.from({ length: count }).map((_, i) => (
156	        <div key={i} className="space-y-2">
157	          <Skeleton className="aspect-[2/3] w-full rounded-2xl" />
158	          <Skeleton className="h-3 w-3/4" />
159	        </div>
160	      ))}
161	    </>
162	  )
163	}
