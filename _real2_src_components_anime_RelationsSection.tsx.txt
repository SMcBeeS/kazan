1	import { useState } from 'react'

4	import { useAsync } from '../../hooks/useAsync'
5	import { anilist } from '../../lib/anilist'
6	import type { JikanRelation } from '../../lib/types'
7	import { cx, relationImage } from '../../lib/utils'
8	import { ErrorState } from '../ui/States'
9	import { Skeleton } from '../ui/Feedback'

12	/**
13	 * RelationsSection â€” "Seasons & Relations" browser.
14	 *
15	 * Collapsible disclosure, closed by default. Only the trigger button is shown
16	 * until the user clicks it; expanding it fetches /anime/{id}/relations and
17	 * renders each relation group (Sequel, Prequel, Side Story, Parent Story, â€¦)
18	 * as a tab so users can jump between seasons cleanly. Each entry links to its
19	 * own detail page.
20	 */
21	export function RelationsSection({ animeId }: { animeId: number }) {
22	  const { t } = useI18n()
23	  const [open, setOpen] = useState(false)
24	
25	  return (
26	    <section className="mt-8">
27	      <button
28	        type="button"
29	        onClick={() => setOpen((o) => !o)}
30	        aria-expanded={open}
31	        aria-controls="relations-panel"
32	        className="flex w-full items-center gap-3 rounded-xl border border-night-700/70 bg-night-850/50 px-4 py-3 text-start transition hover:border-slate-400/60 hover:bg-night-800/70"
33	      >
34	        <span className="flex-1 text-lg font-bold text-white">{t('detail.relationsToggle')}</span>
35	        <ChevronDown
36	          className={cx('h-5 w-5 shrink-0 text-slate-400 transition', open && 'rotate-180')}
37	        />
38	      </button>
39	
40	      {open && (
41	        <div id="relations-panel" className="mt-4">
42	          <p className="mb-4 text-sm text-slate-500">{t('detail.relationsHint')}</p>
43	          <RelationsContent animeId={animeId} />
44	        </div>
45	      )}
46	    </section>
47	  )
48	}
49	
50	/**
51	 * RelationsContent â€” mounts only while the disclosure is open, so the
52	 * relations request is deferred until the user actually asks to see the grid.
53	 */
54	function RelationsContent({ animeId }: { animeId: number }) {
55	  const { t } = useI18n()
56	  const relations = useAsync((s) => anilist.getRelations(animeId, s).then((r) => r.data), [animeId])
57	
58	  if (relations.loading) {
59	    return (
60	      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
61	        {Array.from({ length: 4 }).map((_, i) => (
62	          <Skeleton key={i} className="h-24" />
63	        ))}
64	      </div>
65	    )
66	  }
67	
68	  if (relations.error) {
69	    return (
70	      <ErrorState
71	        message={relations.error}
72	        onRetry={relations.reload}
73	        compact
74	      />
75	    )
76	  }
77	
78	  const groups = (relations.data ?? []).filter((g) => g.entry.length > 0)
79	  if (groups.length === 0) {
80	    return (
81	      <p className="rounded-xl border border-night-700/70 bg-night-850/60 p-4 text-sm text-slate-400">
82	        {t('detail.noRelations')}
83	      </p>
84	    )
85	  }
86	
87	  return <RelationsTabs groups={groups} />
88	}
89	
90	/** Tab switcher over relation groups + grid of related entries. */
91	function RelationsTabs({ groups }: { groups: JikanRelation[] }) {
92	  const [active, setActive] = useState(groups[0]?.relation ?? '')
93	  const current = groups.find((g) => g.relation === active) ?? groups[0]
94	
95	  return (
96	    <div>
97	      {/* Tabs */}
98	      <div className="thin-scrollbar mb-4 flex gap-2 overflow-x-auto pb-1">
99	        {groups.map((g) => (
100	          <button
101	            key={g.relation}
102	            type="button"
103	            onClick={() => setActive(g.relation)}
104	            aria-pressed={g.relation === current.relation}
105	            className={cx(
106	              'shrink-0 rounded-full border px-4 py-1.5 text-xs font-semibold transition',
107	              g.relation === current.relation
108	                ? 'border-white/40 bg-white/10 text-white'
109	                : 'border-night-600/80 bg-night-800/60 text-slate-300 hover:border-slate-400/60 hover:text-white',
110	            )}
111	          >
112	            {g.relation}
113	          </button>
114	        ))}
115	      </div>
116	
117	      {/* Entries */}
118	      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
119	        {current.entry.map((entry) => (
120	          <Link
121	            key={entry.mal_id}
122	            to={`/anime/${entry.mal_id}`}
123	            className="group flex items-center gap-3 rounded-xl border border-night-700/70 bg-night-850/60 p-2 transition hover:border-slate-400/60 hover:bg-night-800/70"
124	          >
125	            <div className="relative h-20 w-14 shrink-0 overflow-hidden rounded-lg bg-night-800">
126	              {relationImage(entry) ? (
127	                <img
128	                  src={relationImage(entry)}
129	                  alt={entry.name}
130	                  loading="lazy"
131	                  className="h-full w-full object-cover transition group-hover:scale-105"
132	                />
133	              ) : (
134	                <div className="flex h-full w-full items-center justify-center text-slate-600">
135	                  ?
136	                </div>
137	              )}
138	              <div className="absolute inset-0 flex items-center justify-center opacity-0 transition group-hover:opacity-100">
139	                <PlayIcon className="h-5 w-5 text-white drop-shadow-lg" />
140	              </div>
141	            </div>
142	            <span className="min-w-0 flex-1">
143	              <span className="line-clamp-3 text-xs font-semibold text-slate-100">{entry.name}</span>
144	            </span>
145	          </Link>
146	        ))}
147	      </div>
148	    </div>
149	  )
150	}
