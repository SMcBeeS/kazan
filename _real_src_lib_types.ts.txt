57	/** Summary item returned by list endpoints (/anime, /top/anime, /seasons/now â€¦) */
58	export interface AnimeSummary {
59	  mal_id: number
60	  url: string
61	  images: { webp?: JikanImage; jpg: JikanImage }
62	  approved?: boolean
63	  titles?: JikanTitle[]
64	  title: string
65	  title_english: string | null
66	  title_japanese: string | null
67	  type: string | null
68	  /**
69	   * Raw AniList format enum (e.g. "TV", "MOVIE", "OVA", "SPECIAL"). Used by the
70	   * UI to special-case movies (runtime instead of episode counts).
71	   */
72	  format?: string | null
73	  source: string
74	  episodes: number | null
75	  status: string | null
76	  airing: boolean
77	  aired: JikanAired
78	  duration: string
79	  rating: string | null
80	  score: number | null
81	  scored_by: number | null
82	  rank: number | null
83	  popularity: number | null
84	  members: number | null
85	  favorites: number | null
86	  synopsis: string | null
87	  season: string | null
88	  year: number | null
89	  studios: JikanStudio[]
90	  genres: JikanGenre[]
91	  themes?: JikanGenre[]
92	  demographics?: JikanGenre[]
93	  /**
94	   * Wide cinematic banner image (AniList `bannerImage`) when available. Preferred
95	   * for hero/backdrop backgrounds; falls back to the poster via utils.bannerImage().
96	   */
97	  bannerImage?: string | null
98	  /**
99	   * Next episode scheduled to air (AniList `nextAiringEpisode`). Used to
100	   * distinguish already-released episodes from upcoming ones.
101	   */
102	  nextAiringEpisode?: { episode: number; airingAt: number | null } | null
103	}

175	/** Generic paginated API envelope (data + pagination) */
176	export interface Paginated<T> {
177	  data: T[]
178	  pagination: {
179	    last_visible_page: number
180	    has_next_page: boolean
181	    current_page: number
182	    items: { count: number; total: number; per_page: number }
183	  }
184	}

186	/** Envelope without pagination (e.g. /genres/anime) */
187	export interface SimpleList<T> {
188	  data: T[]
189	}

191	/* ---------- Video-source domain types (see lib/videoSources.ts) ---------- */
192	
193	export type VideoSourceKind = 'embed' | 'direct'
194	
195	export type VideoQuality = {
196	  /** Human label e.g. "1080p", "720p", "Auto" */
197	  label: string
198	  /** Direct URL for this rendition (only meaningful for `direct` sources) */
199	  url?: string
200	  /** Height in px when known (used for sorting / hls.js level matching) */
201	  height?: number
202	}
203	
204	export type SubtitleTrack = {
205	  /** ISO-ish code, e.g. "ar", "en" */
206	  srclang: string
207	  label: string
208	  /** URL to the subtitle file (.vtt preferred; .srt/.ass converted client-side) */
209	  src: string
210	  /** Original file format for conversion handling */
211	  format: 'vtt' | 'srt' | 'ass'
212	  default?: boolean
213	}
214	
215	export type VideoSource = {
216	  id: string
217	  /** Display name of the provider, e.g. "Server 1", "Primary CDN" */
218	  label: string
219	  kind: VideoSourceKind
220	  /** For embed sources: iframe src. For direct sources: default/highest rendition URL. */
221	  url: string
222	  /** Optional poster override */
223	  poster?: string
224	  /** Available qualities (direct sources); embeds manage their own quality UI */
225	  qualities?: VideoQuality[]
226	  /** Subtitle tracks attached to this source */
227	  subtitles?: SubtitleTrack[]
228	  /** Notes such as licensing/availability reasons, shown to the user */
229	  note?: string
230	}
