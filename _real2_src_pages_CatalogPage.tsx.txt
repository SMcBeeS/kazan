80	/**
81	 * Descending list of recent calendar years for the YEAR filter, anchored at the
82	 * current year (from currentSeason so it always matches the live season).
83	 */
84	function buildYearOptions(tfn: (k: string) => string): { value: string; label: string }[] {
85	  const cur = currentSeason().year
86	  const opts: { value: string; label: string }[] = [{ value: '', label: tfn('catalog.all') }]
87	  for (let y = cur; y >= cur - 15; y--) opts.push({ value: String(y), label: String(y) })
88	  return opts
89	}

91	/**
92	 * CatalogPage â€” search + sort + filter + pagination.
93	 * State lives in the URL (?q=&page=&sort=&order=&type=&status=&genres=&min_score=)
94	 * so searches are shareable and the back button behaves as expected.
95	 */
96	export default function CatalogPage() {
97	  const { t, lang } = useI18n()
98	  const [params, setParams] = useSearchParams()
99	
100	  const q = params.get('q') ?? ''
101	  const page = Math.max(1, parseInt(params.get('page') ?? '1', 10) || 1)
102	  const sort = params.get('sort') ?? 'popularity'
103	  const order = params.get('order') ?? 'desc'
104	  const type = params.get('type') ?? ''
105	  const status = params.get('status') ?? ''
106	  const audience = params.get('audience') ?? ''
107	  const genres = params.get('genres') ?? ''
108	  // Genre NAME deep-link, e.g. /catalog?genre=Action (#5)
109	  const genreNameParam = params.get('genre') ?? ''
110	  const minScore = params.get('min_score') ?? ''
111	  const season = params.get('season') ?? ''
112	  const year = params.get('year') ?? ''
113	
114	  const [searchInput, setSearchInput] = useState(q)
115	  const [showFilters, setShowFilters] = useState(false)
116	
117	  useEffect(() => setSearchInput(q), [q])
118	
119	  // Auto-open the filter panel when a genre deep-link is present so the user
120	  // can see (and clear) the active genre filter.
121	  useEffect(() => {
122	    if (genreNameParam || genres) setShowFilters(true)
123	  }, [genreNameParam, genres])
124	
125	  const commitSearch = useMemo(
126	    () =>
127	      debounce((value: string) => {
128	        setParams(
129	          (prev) => {
130	            const next = new URLSearchParams(prev)
131	            if (value) next.set('q', value)
132	            else next.delete('q')
133	            next.delete('page')
134	            return next
135	          },
136	          { replace: true },
137	        )
138	      }, 300),
139	    [setParams],
140	  )
141	
142	  const update = (key: string, value: string) => {
143	    setParams(
144	      (prev) => {
145	        const next = new URLSearchParams(prev)
146	        if (value) next.set(key, value)
147	        else next.delete(key)
148	        if (key !== 'page') next.delete('page')
149	        return next
150	      },
151	      { replace: true },
152	    )
153	  }
154	
155	  const queryParams: ListParams = useMemo(
156	    () => ({
157	      q: q || undefined,
158	      page,
159	      order_by: sort,
160	      sort: order as 'asc' | 'desc',
161	      type: type || undefined,
162	      status: status || undefined,
163	      tags: AUDIENCE_TAG[audience] || undefined,
164	      genre: genreNameParam || undefined,
165	      genres: genres || undefined,
166	      min_score: minScore ? Number(minScore) : undefined,
167	      season: season || undefined,
168	      season_year: year ? Number(year) : undefined,
169	      sfw: true,
170	      // AniList reports placeholder totals (5000/208) for multi-page result
171	      // sets; ask the adapter to reconcile the REAL total / last page so the
172	      // results badge and pagination are truthful and never reach empty pages.
173	      verify_pagination: true,
174	    }),
175	    [q, page, sort, order, type, status, audience, genreNameParam, genres, minScore, season, year],
176	  )
177	
178	  const result = useAsync((s) => anilist.searchAnime(queryParams, s).then((r) => r), [queryParams])
179	  const genresList = useAsync((s) => anilist.getGenres(s).then((r) => r.data), [])
180	
181	  // Deep links (or stale URLs) can point past the real last page â€” snap back
182	  // instead of showing an empty result set.
183	  useEffect(() => {
184	    const data = result.data
185	    if (!data) return
186	    const lastVisible = data.pagination.last_visible_page
187	    if (data.data.length === 0 && page > 1 && page > lastVisible) {
188	      setParams(
189	        (prev) => {
190	          const next = new URLSearchParams(prev)
191	          next.set('page', String(Math.max(1, lastVisible)))
192	          return next
193	        },
194	        { replace: true },
195	      )
196	    }
197	  }, [result.data, page, setParams])
198	
199	  const totalPages = result.data?.pagination.last_visible_page ?? 1
200	  const totalItems = result.data?.pagination.items.total
201	  // Never display a page number beyond the real last page while clamping.
202	  const displayPage = Math.min(page, Math.max(1, totalPages))
203	
204	  return (
205	    <div className="container-page py-8">
206	      <header className="mb-6">
207	        <h1 className="text-2xl font-semibold tracking-wide text-white sm:text-3xl">{t('catalog.title')}</h1>
208	        <p className="mt-1 text-sm text-slate-400">{t('catalog.subtitle')}</p>
209	      </header>
210	
211	      <SearchBar
212	        value={searchInput}
213	        showFilters={showFilters}
214	        onSearch={(v) => {
215	          setSearchInput(v)
216	          commitSearch(v)
217	        }}
218	        onToggleFilters={() => setShowFilters((v) => !v)}
219	      />
220	
221	      {showFilters && (
222	        <FiltersPanel
223	          values={{ sort, order, type, status, audience, genres, minScore, season, year }}
224	          genres={genresList.data}
225	          onChange={update}
226	          onReset={() => setParams(new URLSearchParams(), { replace: true })}
227	        />
228	      )}
229	
230	      {/* Active genre deep-link chip (#5) */}
231	      {genreNameParam && (
232	        <div className="mb-4 flex items-center gap-2">
233	          <span className="text-sm text-slate-400">{t('catalog.genres')}:</span>
234	          <span className="inline-flex items-center gap-1.5 rounded-full border border-slate-400/50 bg-white/10 px-3 py-1 text-sm font-semibold text-slate-200">
235	            {translateGenre(genreNameParam, lang)}
236	            <button
237	              type="button"
238	              onClick={() => {
239	                setParams(
240	                  (prev) => {
241	                    const next = new URLSearchParams(prev)
242	                    next.delete('genre')
243	                    next.delete('page')
244	                    return next
245	                  },
246	                  { replace: true },
247	                )
248	              }}
249	              aria-label={t('common.close')}
250	              className="text-slate-300 transition hover:text-white"
251	            >
252	              Ã—
253	            </button>
254	          </span>
255	        </div>
256	      )}
257	
258	      <div className="mb-4 flex items-center justify-between text-sm text-slate-400">
259	        <span>
260	          {result.data ? `${t('catalog.results')}: ${totalItems ?? 0}` : t('common.loading')}
261	        </span>
262	        {totalPages > 1 && (
263	          <span>
264	            {t('catalog.page')} {displayPage} {t('catalog.of')} {totalPages}
265	          </span>
266	        )}
267	      </div>
268	
269	      <Results
270	        loading={result.loading}
271	        error={result.error}
272	        items={result.data?.data}
273	        hasData={Boolean(result.data)}
274	        onRetry={result.reload}
275	      />
276	
277	      {result.data && result.data.data.length > 0 && (
278	        <Pagination
279	          page={displayPage}
280	          totalPages={totalPages}
281	          onChange={(p) => {
282	            update('page', String(p))
283	            window.scrollTo({ top: 0, behavior: 'smooth' })
284	          }}
285	        />
286	      )}
287	    </div>
288	  )
289	}

293	function SearchBar({
294	  value,
295	  showFilters,
296	  onSearch,
297	  onToggleFilters,
298	}: {
299	  value: string
300	  showFilters: boolean
301	  onSearch: (v: string) => void
302	  onToggleFilters: () => void
303	}) {
304	  const { t } = useI18n()
305	  return (
306	    <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center">
307	      <div className="relative flex-1">
308	        <SearchIcon className="pointer-events-none absolute start-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-500" />
309	        <input
310	          type="search"
311	          value={value}
312	          onChange={(e) => onSearch(e.target.value)}
313	          placeholder={t('catalog.searchPlaceholder')}
314	          aria-label={t('catalog.searchPlaceholder')}
315	          className="w-full rounded-xl border border-night-700/70 bg-night-850 py-3 pe-3 ps-9 text-sm text-slate-100 placeholder:text-slate-500 focus:border-white/40"
316	        />
317	      </div>
318	      <button onClick={onToggleFilters} className="btn-ghost" aria-expanded={showFilters}>
319	        <FilterIcon className="h-4 w-4" />
320	        {showFilters ? t('catalog.hideFilters') : t('catalog.showFilters')}
321	      </button>
322	    </div>
323	  )
324	}

328	function FiltersPanel({
329	  values,
330	  genres,
331	  onChange,
332	  onReset,
333	}: {
334	  values: {
335	    sort: string
336	    order: string
337	    type: string
338	    status: string
339	    audience: string
340	    genres: string
341	    season: string
342	    year: string
343	    minScore: string
344	  }
345	  genres?: { mal_id: number; name: string }[] | null
346	  onChange: (key: string, value: string) => void
347	  onReset: () => void
348	}) {
349	  const { t, lang } = useI18n()
350	  // Localised option lists for format/status (Arabic â‡„ English).
351	  const typeOptions = [
352	    { value: '', label: t('catalog.all') },
353	    ...TYPES.map((x) => ({ value: x, label: t(typeLabelKey[x]) })),
354	  ]
355	  const statusOptions = [
356	    { value: '', label: t('catalog.all') },
357	    ...STATUSES.map((x) => ({ value: x, label: t(statusLabelKey[x]) })),
358	  ]
359	  // Keep unknown deep-linked values (e.g. legacy `?type=ona`) selectable.
360	  if (values.type && !typeOptions.some((o) => o.value === values.type)) {
361	    typeOptions.push({ value: values.type, label: values.type.toUpperCase() })
362	  }
363	  if (values.status && !statusOptions.some((o) => o.value === values.status)) {
364	    statusOptions.push({ value: values.status, label: values.status })
365	  }
366	  return (
367	    <div className="mb-6 grid grid-cols-2 gap-3 rounded-2xl border border-night-700/70 bg-night-850/60 p-4 sm:grid-cols-3 lg:grid-cols-6">
368	      {/* YEAR + SEASON â€” the primary "Browse Anime" filters, rendered like
369	          the other grid inputs (no extra wrapper box). */}
370	      <Select
371	        label={t('catalog.year')}
372	        value={values.year}
373	        onChange={(v) => onChange('year', v)}
374	        options={buildYearOptions(t)}
375	      />
376	      <Select
377	        label={t('catalog.season')}
378	        value={values.season}
379	        onChange={(v) => onChange('season', v)}
380	        options={[
381	          { value: '', label: t('catalog.all') },
382	          ...SEASONS.map((s) => ({ value: s, label: t(seasonLabelKey[s]) })),
383	        ]}
384	      />
385	      <Select
386	        label={t('catalog.sort')}
387	        value={values.sort}
388	        onChange={(v) => onChange('sort', v)}
389	        options={SORTS.map((s) => ({ value: s, label: t(sortLabelKey[s]) }))}
390	      />
391	      <Select
392	        label={t('catalog.order')}
393	        value={values.order}
394	        onChange={(v) => onChange('order', v)}
395	        options={[
396	          { value: 'desc', label: t('catalog.desc') },
397	          { value: 'asc', label: t('catalog.asc') },
398	        ]}
399	      />
400	      <Select
401	        label={t('catalog.type')}
402	        value={values.type}
403	        onChange={(v) => onChange('type', v)}
404	        options={typeOptions}
405	      />
406	      <Select
407	        label={t('catalog.status')}
408	        value={values.status}
409	        onChange={(v) => onChange('status', v)}
410	        options={statusOptions}
411	      />
412	      <Select
413	        label={t('catalog.audience')}
414	        value={values.audience}
415	        onChange={(v) => onChange('audience', v)}
416	        options={AUDIENCES.map((x) => ({ value: x, label: t(audienceLabelKey[x]) }))}
417	      />
418	      <TextInput
419	        label={t('catalog.minScore')}
420	        type="number"
421	        value={values.minScore}
422	        onChange={(v) => onChange('min_score', v)}
423	        placeholder="0"
424	      />
425	      {genres && (
426	        <Select
427	          label={t('catalog.genres')}
428	          value={values.genres}
429	          onChange={(v) => onChange('genres', v)}
430	          options={[
431	            { value: '', label: 'â€”' },
432	            ...genres.map((g) => ({ value: String(g.mal_id), label: translateGenre(g.name, lang) })),
433	          ]}
434	        />
435	      )}
436	      <div className="col-span-full flex justify-center mt-2">
437	        <button onClick={onReset} className="btn-ghost w-48">
438	          {t('catalog.reset')}
439	        </button>
440	      </div>
441	    </div>
442	  )
443	}

445	/* -------------------------------- results -------------------------------- */
446	
447	function Results({
448	  loading,
449	  error,
450	  items,
451	  hasData,
452	  onRetry,
453	}: {
454	  loading: boolean
455	  error: string | null
456	  items?: AnimeSummary[]
457	  hasData: boolean
458	  onRetry: () => void
459	}) {
460	  const { t } = useI18n()
461	  if (loading) {
462	    return (
463	      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6">
464	        <AnimeCardSkeleton count={18} />
465	      </div>
466	    )
467	  }
468	  if (error && !hasData) return <ErrorState message={error} onRetry={onRetry} />
469	  if (items && items.length === 0) {
470	    return <EmptyState title={t('catalog.noResults')} hint={t('catalog.noResultsHint')} />
471	  }
472	  return (
473	    <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6">
474	      {items?.map((a) => (
475	        <AnimeCard key={a.mal_id} anime={a} />
476	      ))}
477	    </div>
478	  )
479	}
