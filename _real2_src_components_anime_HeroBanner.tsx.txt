3	import { useEffect, useRef, useState } from 'react'
4	import type { CSSProperties } from 'react'
5	import { Link } from 'react-router-dom'
6	import type { AnimeSummary } from '../../lib/types'
7	import { useI18n } from '../../context/LanguageContext'
8	import { coverImage, cx } from '../../lib/utils'
9	import { ArrowLeft, ArrowRight, PlayIcon } from '../icons'
10	import { Skeleton } from '../ui/Feedback'
11	
12	/**
13	 * Shared GPU-compositing hint for the crossfading slide layers:
14	 * `will-change: opacity` promotes each layer to its own compositor layer,
15	 * and `transform: translateZ(0)` forces GPU compositing so the 700ms
16	 * crossfade never triggers layout/paint on the main thread.
17	 */
18	const SLIDE_LAYER_STYLE: CSSProperties = {
19	  willChange: 'opacity',
20	  transform: 'translateZ(0)',
21	  backfaceVisibility: 'hidden',
22	}
23	
24	/** How long each slide stays visible before auto-advancing (#1). */
25	const SLIDE_INTERVAL_MS = 7000
26	
27	/** Banners narrower than this are treated as low-res and never stretched. */
28	
29	/**
30	 * HeroBanner â€” auto-rotating cinematic carousel spotlighting the top
31	 * seasonal anime (top 5â€“8 items): smooth crossfade transitions, auto-advance
32	 * every 7 seconds, left/right arrows, bottom dots, pause-on-hover (#1).
33	 */
34	export function HeroBanner({ items }: { items: AnimeSummary[] }) {
35	  const { t } = useI18n()
36	  const [index, setIndex] = useState(0)
37	  const [paused, setPaused] = useState(false)
38	  // Whether the banner is actually on screen. The 7s auto-advance timer is
39	  // paused while it is scrolled out of view so hidden slides never churn.
40	  const sectionRef = useRef<HTMLElement | null>(null)
41	  const [inView, setInView] = useState(true)
42	  // Mal ids whose wide banner measured below the high-res threshold on load.
43	  const [lowResBanners, setLowResBanners] = useState<Set<number>>(() => new Set())
44	  const markLowRes = (malId: number, width: number) => {
45	    if (width >= 1280) return
46	    setLowResBanners((prev) => {
47	      if (prev.has(malId)) return prev
48	      const next = new Set(prev)
49	      next.add(malId)
50	      return next
51	    })
52	  }
53	
54	  const count = items.length
55	
56	  // Keep the active index in range if the list shrinks.
57	  useEffect(() => {
58	    if (index >= count) setIndex(0)
59	  }, [count, index])
60	
61	  // Stop the auto-advance timer entirely when the banner scrolls out of view.
62	  useEffect(() => {
63	    const el = sectionRef.current
64	    if (!el || typeof IntersectionObserver === 'undefined') return
65	    const obs = new IntersectionObserver(
66	      (entries) => setInView(entries.some((e) => e.isIntersecting)),
67	      { threshold: 0.25 },
68	    )
69	    obs.observe(el)
70	    return () => obs.disconnect()
71	  }, [count])
72	
73	  // Auto-advance while not hovered/focused AND actually visible.
74	  // The interval is cleared on unmount and whenever the banner leaves the
75	  // viewport, so no background slide re-renders happen off-screen.
76	  useEffect(() => {
77	    if (count <= 1 || paused || !inView) return
78	    const id = window.setInterval(() => {
79	      setIndex((i) => (i + 1) % count)
80	    }, SLIDE_INTERVAL_MS)
81	    return () => window.clearInterval(id)
82	  }, [count, paused, inView])
83	
84	  if (count === 0) {
85	    return (
86	      <div className="container-page pt-5">
87	        <div className="relative h-[54vh] min-h-[380px] w-full overflow-hidden rounded-3xl bg-night-900">
88	          <Skeleton className="h-full w-full rounded-none" />
89	        </div>
90	      </div>
91	    )
92	  }
93	
94	  const active = Math.min(index, count - 1)
95	  const anime = items[active]
96	  const go = (next: number) => setIndex((next + count) % count)
97	
98	  const title = anime.title_english || anime.title
99	  const synopsis = anime.synopsis
100	    ? anime.synopsis.length > 240
101	      ? anime.synopsis.slice(0, 240).trimEnd() + 'â€¦'
102	      : anime.synopsis
103	    : ''
104	
105	  return (
106	    <div className="w-full pt-5">
107	    <div className="container-page relative">
108	    <section
109	      ref={sectionRef}
110	      className="relative isolate h-[54vh] min-h-[380px] w-full overflow-hidden rounded-3xl shadow-card ring-1 ring-night-700/60"
111	      onMouseEnter={() => setPaused(true)}
112	      onMouseLeave={() => setPaused(false)}
113	      onFocus={() => setPaused(true)}
114	      onBlur={() => setPaused(false)}
115	      aria-roledescription="carousel"
116	    >
117	      {/*
118	        Crossfading backgrounds â€” every slide fills the container edge-to-edge.
119	        A wide bannerImage renders with object-cover; when a banner is missing
120	        or narrow, the container is filled with a blurred scale-110 crop of the
121	        coverImage and the crisp poster artwork is placed gracefully to the side.
122	        No empty black pillarbars are ever rendered.
123	      */}
124	      {items.map((a, i) => {
125	        const banner = a.bannerImage || ''
126	        const lowRes = lowResBanners.has(a.mal_id)
127	        const wide = Boolean(banner) && !lowRes
128	        const cover = coverImage(a)
129	        const src = wide ? banner : (cover || banner)
130	        if (!src) return null
131	        const loading = i === 0 ? 'eager' : 'lazy'
132	        return (
133	          <div
134	            key={a.mal_id}
135	            aria-hidden="true"
136	            style={SLIDE_LAYER_STYLE}
137	            className={cx(
138	              "absolute inset-0 transition-opacity duration-700 ease-in-out",
139	              i === active ? "opacity-100" : "opacity-0",
140	            )}
141	          >
142	            {/* Ambient blurred backdrop Ã¢â‚¬â€ always fills edge-to-edge */}
143	            <img
144	              src={wide ? banner : cover}
145	              alt=""
146	              loading={loading}
147	              decoding="async"
148	              className="absolute inset-0 h-full w-full scale-110 object-cover object-center blur-2xl opacity-40"
149	            />
150	            {wide ? (
151	              <img
152	                src={src}
153	                alt=""
154	                loading={loading}
155	                decoding="async"
156	                onLoad={(e) => markLowRes(a.mal_id, e.currentTarget.naturalWidth)}
157	                className="absolute inset-0 h-full w-full object-cover object-center"
158	              />
159	            ) : (
160	              <img
161	                src={src}
162	                alt=""
163	                loading={loading}
164	                decoding="async"
165	                className="relative block h-full max-w-[60%] object-contain object-right"
166	              />
167	            )}
168	          </div>
169	        )
170	      })}
171	
172	      <div className="absolute inset-0 bg-gradient-to-t from-[#030712]/80 via-[#030712]/40 to-transparent" />
173	      <div className="absolute inset-0 bg-gradient-to-r from-[#030712] via-[#030712]/60 to-transparent rtl:bg-gradient-to-l" />
174	
175	      {/* Content â€” keyed so the fade-up animation replays per slide */}
176	      <div className="relative flex h-full flex-col justify-end px-5 pb-16 pt-24 sm:px-8 sm:pb-16">
177	        <div key={anime.mal_id} className="max-w-2xl animate-fade-up">
178	          <p className="text-[11px] font-bold uppercase tracking-[0.25em] text-slate-300">
179	            {t('hero.featured')}
180	          </p>
181	
182	          <h2 className="mt-2 text-3xl font-semibold leading-tight tracking-wide text-white text-shadow-lg sm:text-5xl">
183	            {title}
184	          </h2>
185	
186	          {synopsis && (
187	            <p className="mt-3 line-clamp-2 max-w-xl text-sm leading-relaxed text-slate-300 sm:text-base">
188	              {synopsis}
189	            </p>
190	          )}
191	
192	          <div className="mt-6 flex flex-wrap items-center gap-3">
193	            <Link
194	              to={`/watch/${anime.mal_id}/1`}
195	              className="btn-primary rounded-full px-7 py-3 text-base"
196	            >
197	              <PlayIcon className="h-5 w-5" />
198	              {t('hero.watchNow')}
199	            </Link>
200	            <Link
201	              to={`/anime/${anime.mal_id}`}
202	              className="inline-flex items-center justify-center gap-2 rounded-full border border-white/25 bg-white/10 px-6 py-3 text-base font-semibold text-white backdrop-blur-md transition hover:border-white/40 hover:bg-white/20 active:scale-[0.98]"
203	            >
204	              {t('hero.moreInfo')}
205	            </Link>
206	          </div>
207	        </div>
208	      </div>
209	
210	      {/* Bottom pill indicators */}
211	      {count > 1 && (
212	        <div className="absolute inset-x-0 bottom-5 z-10 flex items-center justify-center gap-1.5">
213	          {items.map((a, i) => (
214	            <button
215	              key={a.mal_id}
216	              type="button"
217	              aria-label={`${t('hero.goToSlide')} ${i + 1}`}
218	              aria-current={i === active ? 'true' : undefined}
219	              onClick={() => go(i)}
220	              className={cx(
221	                'h-1.5 rounded-full transition-all duration-300',
222	                i === active ? 'w-8 bg-white shadow' : 'w-3 bg-white/35 hover:bg-white/70',
223	              )}
224	            />
225	          ))}
226	        </div>
227	      )}
228	    </section>
229	
230	    {/* Navigation arrows — placed outside the card, in the side void */}
231	    {count > 1 && (
232	      <>
233	        <button
234	          type="button"
235	          aria-label={t('hero.prev')}
236	          onClick={() => go(active - 1)}
237	          className="absolute z-10 top-1/2 flex h-11 w-11 -translate-y-1/2 items-center justify-center rounded-full border border-white/15 bg-night-950/60 text-slate-200 backdrop-blur-sm transition hover:border-slate-400/60 hover:text-white -start-3"
238	        >
239	          <ArrowLeft className="h-5 w-5 flip-rtl" />
240	        </button>
241	        <button
242	          type="button"
243	          aria-label={t('hero.next')}
244	          onClick={() => go(active + 1)}
245	          className="absolute z-10 top-1/2 flex h-11 w-11 -translate-y-1/2 items-center justify-center rounded-full border border-white/15 bg-night-950/60 text-slate-200 backdrop-blur-sm transition hover:border-slate-400/60 hover:text-white -end-3"
246	        >
247	          <ArrowRight className="h-5 w-5 flip-rtl" />
248	        </button>
249	      </>
250	    )}
251	    </div>
252	    </div>
253	  )
254	}
